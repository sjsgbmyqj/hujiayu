📚 这一页的 fetch 是怎么写的

列表页只做三件事：请求列表 → 循环渲染卡片 → 把商品 id 拼进详情页链接。 点击卡片时并不会再发请求，真正的详情数据由详情页自己去取（按需请求）。 

// 1. 请求列表接口（跨域，接口返回 { products: [...], total, skip, limit }）
var res = await fetch('https://dummyjson.com/products?limit=8');
if (!res.ok) throw new Error('HTTP ' + res.status);
var data = await res.json();

// 2. 取出数组，循环生成卡片
data.products.forEach(function (item) {
    var a = document.createElement('a');
    a.href = './17-3-detail.html?id=' + item.id;   // 关键：用 id 传参
    a.innerHTML = '<img src="' + item.thumbnail + '">' + item.title;
    document.getElementById('hjy_grid').appendChild(a);
});

// 3. 筛选 / 排序都在本地对 data.products 做，不用重新请求
