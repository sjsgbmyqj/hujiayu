⑤ 这么写就够用了

async / await 更像同步代码，推荐
async function hjy_load() {
    try {
        var res = await fetch(
            'https://jsonplaceholder.typicode.com/users'
        );

        // 404 / 500 不会让 fetch 报错，必须自己判断
        if (!res.ok) {
            throw new Error('HTTP ' + res.status);
        }

        var users = await res.json();  // 解析成 JS 数组
        hjy_render(users);             // 渲染到页面
    } catch (err) {
        console.error('请求失败：', err.message);
    }
}
Promise .then() 链式写法，理解 Promise 的基础
fetch('https://jsonplaceholder.typicode.com/users')
    .then(function (res) {
        if (!res.ok) {
            throw new Error('HTTP ' + res.status);
        }
        return res.json();            // 返回 Promise
    })
    .then(function (users) {
        // users 是 10 个用户对象组成的数组
        hjy_render(users);
    })
    .catch(function (err) {
        console.error('请求失败：', err.message);
    });

四个要点：① fetch() 返回 Promise，要 await 或 .then() 拿结果； ② res.ok 为 false 时要自己 throw，否则会把错误页面当成正常数据； ③ 数据只能用一次 res.json()，解析后存成变量，反复渲染都用这个变量； ④ 断网或跨域被拦会抛 TypeError: Failed to fetch，要用 try/catch 兜住并提示用户。
